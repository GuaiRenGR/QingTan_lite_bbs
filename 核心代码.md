## 6. 核心 Service

### 6.1 积分服务

`app/Services/ScoreService.php`

```php
<?php

namespace App\Services;

use App\Models\ScoreLog;
use App\Models\User;
use Illuminate\Support\Facades\DB;

class ScoreService
{
    public function change(User $user, int $amount, string $action, string $remark = ''): User
    {
        return DB::transaction(function () use ($user, $amount, $action, $remark) {
            $user = User::whereKey($user->id)->lockForUpdate()->firstOrFail();

            $user->score += $amount;

            if ($user->score < 0) {
                $user->score = 0;
            }

            $user->level = $this->calculateLevel($user->score);
            $user->save();

            ScoreLog::create([
                'user_id' => $user->id,
                'action' => $action,
                'score_type' => 'score',
                'amount' => $amount,
                'balance' => $user->score,
                'remark' => $remark,
            ]);

            return $user;
        });
    }

    public function calculateLevel(int $score): int
    {
        return match (true) {
            $score >= 10000 => 6,
            $score >= 3000 => 5,
            $score >= 1000 => 4,
            $score >= 500 => 3,
            $score >= 100 => 2,
            default => 1,
        };
    }
}
```

### 6.2 发帖服务

`app/Services/ThreadService.php`

```php
<?php

namespace App\Services;

use App\Models\Forum;
use App\Models\Thread;
use App\Models\User;
use Illuminate\Support\Facades\DB;
use Illuminate\Validation\ValidationException;

class ThreadService
{
    public function __construct(
        protected ScoreService $scoreService
    ) {}

    public function create(User $user, array $data): Thread
    {
        if (!$user->isNormal()) {
            throw ValidationException::withMessages([
                'user' => '账号状态异常，无法发帖',
            ]);
        }

        $forum = Forum::where('status', 1)->findOrFail($data['forum_id']);

        return DB::transaction(function () use ($user, $forum, $data) {
            $thread = Thread::create([
                'forum_id' => $forum->id,
                'user_id' => $user->id,
                'category_id' => $data['category_id'] ?? null,
                'type' => $data['type'] ?? 'normal',
                'title' => $data['title'],
                'summary' => mb_substr(strip_tags($data['content']), 0, 120),
                'content' => $data['content'],
                'cover' => $data['cover'] ?? null,
                'tags' => $data['tags'] ?? [],
                'status' => Thread::STATUS_NORMAL,
                'last_reply_at' => now(),
            ]);

            $forum->increment('thread_count');
            $forum->increment('today_count');

            $this->scoreService->change($user, 5, 'create_thread', '发布主题奖励');

            return $thread->load(['user:id,nickname,avatar', 'forum:id,name']);
        });
    }
}
```

### 6.3 回复服务

`app/Services/PostService.php`

```php
<?php

namespace App\Services;

use App\Models\Post;
use App\Models\Thread;
use App\Models\User;
use Illuminate\Support\Facades\DB;
use Illuminate\Validation\ValidationException;

class PostService
{
    public function __construct(
        protected ScoreService $scoreService
    ) {}

    public function create(User $user, Thread $thread, array $data): Post
    {
        if (!$user->isNormal()) {
            throw ValidationException::withMessages([
                'user' => '账号状态异常，无法回复',
            ]);
        }

        if ($thread->is_closed) {
            throw ValidationException::withMessages([
                'thread' => '主题已关闭，无法回复',
            ]);
        }

        return DB::transaction(function () use ($user, $thread, $data) {
            $floor = Post::where('thread_id', $thread->id)->max('floor') + 1;

            $post = Post::create([
                'thread_id' => $thread->id,
                'user_id' => $user->id,
                'parent_id' => $data['parent_id'] ?? null,
                'quote_post_id' => $data['quote_post_id'] ?? null,
                'content' => $data['content'],
                'floor' => $floor,
                'status' => 1,
            ]);

            $thread->increment('reply_count');
            $thread->forceFill([
                'last_reply_at' => now(),
            ])->save();

            $thread->forum()->increment('post_count');

            $this->scoreService->change($user, 2, 'create_post', '发布回复奖励');

            return $post->load('user:id,nickname,avatar');
        });
    }
}
```

### 6.4 签到服务

`app/Services/CheckinService.php`

```php
<?php

namespace App\Services;

use App\Models\Checkin;
use App\Models\User;
use Illuminate\Support\Facades\DB;
use Illuminate\Validation\ValidationException;

class CheckinService
{
    public function __construct(
        protected ScoreService $scoreService
    ) {}

    public function status(User $user): array
    {
        $today = now()->toDateString();

        $todayRecord = Checkin::where('user_id', $user->id)
            ->where('checkin_date', $today)
            ->first();

        $latest = Checkin::where('user_id', $user->id)
            ->latest('checkin_date')
            ->first();

        return [
            'checked_today' => (bool) $todayRecord,
            'continuous_days' => $latest?->continuous_days ?? 0,
            'today_reward' => $this->calculateReward(($latest?->continuous_days ?? 0) + 1),
        ];
    }

    public function checkin(User $user): Checkin
    {
        $today = now()->toDateString();
        $yesterday = now()->subDay()->toDateString();

        return DB::transaction(function () use ($user, $today, $yesterday) {
            $exists = Checkin::where('user_id', $user->id)
                ->where('checkin_date', $today)
                ->lockForUpdate()
                ->exists();

            if ($exists) {
                throw ValidationException::withMessages([
                    'checkin' => '今日已签到',
                ]);
            }

            $yesterdayRecord = Checkin::where('user_id', $user->id)
                ->where('checkin_date', $yesterday)
                ->first();

            $continuousDays = $yesterdayRecord
                ? $yesterdayRecord->continuous_days + 1
                : 1;

            $reward = $this->calculateReward($continuousDays);

            $record = Checkin::create([
                'user_id' => $user->id,
                'checkin_date' => $today,
                'reward_score' => $reward,
                'continuous_days' => $continuousDays,
            ]);

            $this->scoreService->change($user, $reward, 'checkin', "连续签到 {$continuousDays} 天");

            return $record;
        });
    }

    private function calculateReward(int $continuousDays): int
    {
        if ($continuousDays >= 30) {
            return 100;
        }

        if ($continuousDays >= 7) {
            return 20;
        }

        return 5 + min($continuousDays - 1, 5);
    }
}
```

```dart
    this.user,

    Map<String, dynamic>? user,

  Future<void> login({
    required String account,
    required String password,
  }) async {
    state = state.copyWith(loading: true);

    try {
      final res = await ApiClient.instance.dio.post('/auth/login', data: {
        'account': account,
        'password': password,
      });

      final data = res.data['data'];
      final token = data['access_token'];

      await ApiClient.instance.storage.write(
        key: 'access_token',
        value: token,
      );

      state = AuthState(
        loading: false,
        loggedIn: true,
        user: data['user'],
      );
    } finally {
      state = state.copyWith(loading: false);
    }
  }

  Future<void> logout() async {

      await ApiClient.instance.dio.post('/auth/logout');
    } catch (_) {}

    state = AuthState(
      loading: false,
      loggedIn: false,
      user: null,
    );
  }

    final token = await ApiClient.instance.storage.read(key: 'access_token');

    if (token == null) {
      state = state.copyWith(loggedIn: false);
      return;

      );
```

```dart
  });

  FeedState copyWith({
    bool? loading,
    bool? loadingMore,
    bool? noMore,
    int? page,
    List<ThreadModel>? items,
  }) {
    return FeedState(
      loading: loading ?? this.loading,
      loadingMore: loadingMore ?? this.loadingMore,
      noMore: noMore ?? this.noMore,
      page: page ?? this.page,
      items: items ?? this.items,
    );
  }
}

class FeedController extends StateNotifier<FeedState> {
  final FeedRepository repository;

  FeedController(this.repository) : super(FeedState());

  Future<void> refresh() async {
    state = state.copyWith(loading: true, page: 1, noMore: false);

    try {
      final items = await repository.fetchFeed(page: 1);

      state = state.copyWith(
        loading: false,
        items: items,
        page: 1,
        noMore: items.isEmpty,
      );
    } finally {
      state = state.copyWith(loading: false);
    }
  }

  Future<void> loadMore() async {
    if (state.loadingMore || state.noMore) return;

    state = state.copyWith(loadingMore: true);

    final nextPage = state.page + 1;

    try {
      final items = await repository.fetchFeed(page: nextPage);

      state = state.copyWith(
        loadingMore: false,
        page: nextPage,
        items: [...state.items, ...items],
        noMore: items.isEmpty,
      );
    } finally {
      state = state.copyWith(loadingMore: false);
    }
  }
}
```

```dart
class ThreadRepository {

    );

    return res.data['data']['data'];
  }

  Future<void> reply({
    required int threadId,
    required String content,

    await ApiClient.instance.dio.post(
      '/threads/$threadId/posts',
      data: {
        'content': content,
      },
    );
  }

  Future<void> like(int threadId) async {
    await ApiClient.instance.dio.post('/threads/$threadId/like');
  }

  Future<void> favorite(int threadId) async {
    await ApiClient.instance.dio.post('/threads/$threadId/favorite');
  }
```
